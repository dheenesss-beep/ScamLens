const SAMPLES = [
  {label:"Courier hold", text:"India Post: Your parcel is held at our warehouse due to incomplete address. Update within 12 hours or it will be returned: https://indiapost-redelivery.top/track"},
  {label:"KYC expiry", text:"Dear Customer, your SBI YONO account will be blocked today. Complete KYC immediately: http://sbi-kyc-verify.xyz/login"},

  {label:"Lookalike link", text:"Your Paytm wallet is suspended. Re-verify your account at https://paytrn-secure.com/verify to avoid permanent closure."},
  {label:"Task job", text:"Part time job! Earn ₹3000-8000 daily from home by liking YouTube videos. Contact HR on WhatsApp: wa.me/919812345678"},
  {label:"Bank debit alert", legit:true, text:"Rs 450.00 debited from A/c XX1234 on 04-10-26 to VPA swiggy@icici. Not you? Call 18002586161 - HDFC Bank"},
  {label:"Amazon shipping", legit:true, text:"Your Amazon order #405-1182934 has shipped and arrives Thursday. Track it at https://www.amazon.in/gp/your-account/order-history"}
];

function esc(s){
  return String(s).replace(/[&<>\"']/g, c => ({
    "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
  }[c]));
}

async function analyzeWithBackend(text){
  const response = await fetch("/api/analyze", {
    method: "POST",
    headers: {"Content-Type":"application/json"},
    body: JSON.stringify({text})
  });

  if (!response.ok){
    let detail = "The ScamLens backend could not analyse the message.";
    try {
      const data = await response.json();
      if (data.error) detail = data.error;
    } catch(e){}
    throw new Error(detail);
  }
  return response.json();
}

function render(r){
  const el = document.getElementById("result");
  const level = r.score >= 60 ? "high" : r.score >= 25 ? "med" : "low";
  const color = level==="high" ? "var(--danger)" : level==="med" ? "var(--warn)" : "var(--safe)";
  const head = {high:"High risk",med:"Suspicious",low:"Looks safe"}[level];
  const sub = {
    high:"Strong scam signals. Do not click, call or pay.",
    med:"Some red flags. Verify through the official app or website first.",
    low:"No major red flags found. Stay alert anyway."
  }[level];

  const advice = {
    high:"Don't tap the link or reply. Block the sender and report it through the appropriate official cybercrime channel. If money was lost in India, call 1930.",
    med:"Open the company's official app or type its website yourself. Never use the link or number in the message.",
    low:"This looks like a normal message. Still, never share an OTP or PIN with anyone."
  }[level];

  const sevLabel = {high:"High",med:"Medium",info:"Note",good:"Safe"};
  const order = {high:0,med:1,info:2,good:3};
  const flags = [...(r.flags || [])].sort((a,b)=>order[a.sev]-order[b.sev]);
  const redFlags = flags.filter(f => f.sev !== "good");

  el.innerHTML = `
    <div class="pane-label">Verdict</div>
    <div class="verdict">
      <div class="lens" style="--c:${color};background:conic-gradient(${color} ${r.score}%, var(--line) 0)">
        <div><span><b>${r.score}</b><small>risk</small></span></div>
      </div>
      <div class="vtext"><h3 style="color:${color}">${head}</h3><p>${sub}</p></div>
    </div>

    ${r.qr ? `<div class="link"><span class="hint">QR decoded:</span> ${esc(r.qr)}</div>` : ""}

    ${flags.length ? `
      <div class="pane-label">${redFlags.length} red flag${redFlags.length===1?"":"s"} found</div>
      <ul class="flags">
        ${flags.map(f => `
          <li class="flag">
            <span class="sev ${f.sev}">${sevLabel[f.sev]}</span>
            <div>
              <strong>${esc(f.title)}</strong>
              <span class="d">${esc(f.detail)}</span>
            </div>
          </li>
        `).join("")}
      </ul>
    ` : `<p class="empty">No red flags matched.</p>`}

    ${r.links && r.links.length ? `
      <div class="pane-label">Links found</div>
      <div class="links">
        ${r.links.map(l => `
          <div class="link">
            <span class="host">${esc(l.host)}</span>${esc(l.path || "")}
            ${l.official ? " · official" : (l.notes && l.notes.length ? " · " + esc(l.notes.join(", ")) : "")}
          </div>
        `).join("")}
      </div>
    ` : ""}

    <div class="advice ${level}">${advice}</div>
  `;
}

async function runScan(textOverride=null){
  const msg = document.getElementById("msg");
  const text = (textOverride !== null ? textOverride : msg.value).trim();

  if (!text){
    document.getElementById("result").innerHTML =
      `<div class="pane-label">Verdict</div><p class="empty">Paste a message or link, then press Scan message.</p>`;
    return;
  }

  document.getElementById("result").innerHTML =
    `<div class="pane-label">Verdict</div><p class="empty">Analysing with ScamLens…</p>`;

  try{
    const result = await analyzeWithBackend(text);
    render(result);
  }catch(error){
    document.getElementById("result").innerHTML =
      `<div class="pane-label">Error</div><p class="empty">${esc(error.message)}</p>`;
  }
}

  const b = document.createElement("button");

  b.textContent=s.label;
  b.addEventListener("click",()=>{
    document.getElementById("msg").value = s.text;

  chipBox.appendChild(b);
});

  }

document.getElementById("qrfile").addEventListener("change", e=>{
  const f = e.target.files[0];
  if (!f) return;

  const reader = new FileReader();

  reader.onload = () => {
    const img = new Image();

    img.onload = async () => {
      const c = document.createElement("canvas");
      const scale = Math.min(1, 1200 / Math.max(img.width, img.height));

      c.width = Math.round(img.width * scale);
      c.height = Math.round(img.height * scale);

      const ctx = c.getContext("2d");
      ctx.drawImage(img, 0, 0, c.width, c.height);

      const d = ctx.getImageData(0, 0, c.width, c.height);
      const code = window.jsQR ? jsQR(d.data, c.width, c.height) : null;

      if (!code){
        document.getElementById("result").innerHTML =
          `<div class="pane-label">Verdict</div><p class="empty">No QR code found in that image. Try a sharper, closer photo of the code.</p>`;
        return;
      }

      document.getElementById("result").innerHTML =
        `<div class="pane-label">Verdict</div><p class="empty">QR decoded. Analysing…</p>`;

      try{
        const result = await analyzeWithBackend(code.data);
        result.qr = code.data;
        render(result);
      }catch(error){
        document.getElementById("result").innerHTML =
          `<div class="pane-label">Error</div><p class="empty">${esc(error.message)}</p>`;
      }
    };

    img.src = reader.result;
  };

  reader.readAsDataURL(f);
  e.target.value = "";
});

// Open in a working state.
document.getElementById("msg").value = SAMPLES[1].text;
runScan();
